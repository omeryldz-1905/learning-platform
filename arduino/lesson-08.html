Ders 8: DHT11 Sıcaklık ve Nem Sensörü

Orta

DHT11 sensörü kullanarak ortamın sıcaklık ve nem oranını ölçeceğiz. Bu sensör hava durumu istasyonları ve iklim kontrol sistemlerinde kullanılır.

📚 Öğrenecekleriniz

  • DHT11 sensörün çalışma prensibi
  • Digital protokol ile veri okuma
  • Sıcaklık (°C) ve nem (%) ölçümü
  • Seri monitöre veri gönderme
  • Kalibrasyon ve hata kontrolü
  • Grafik arayüzle gerçek zamanlı izleme
  • Veri günlüğe kaydetme (logging)

DHT11 Pin Dağılımı:
1: VCC (5V) - Güç
2: DATA - Veri Hattı (Pin 2)
3: NC - Bağlı değil
4: GND - Toprak

💻 Arduino Kodu (Temel Versiyon)

#include 

#define DHTPIN 2      // DHT11 veri pini
#define DHTTYPE DHT11 // DHT 11

DHT dht(DHTPIN, DHTTYPE);

void setup() {
  Serial.begin(9600);
  Serial.println("DHT11 Sensör Test Başladı...");
  dht.begin();
}

void loop() {
  // 2 saniye bekle (DHT11 minimum okuma süresi)
  delay(2000);

  // Nem ve sıcaklık oku
  float nem = dht.readHumidity();
  float sicaklik = dht.readTemperature();

  // Hata kontrol
  if (isnan(nem) || isnan(sicaklik)) {
    Serial.println("DHT11 okunamadı!");
    return;
  }

  // Hissi sıcaklık hesapla (Heat Index)
  float hissi = dht.computeHeatIndex(sicaklik, nem, false);

  // Seri monitöre yaz
  Serial.print("Nem: ");
  Serial.print(nem);
  Serial.print("%  Sıcaklık: ");
  Serial.print(sicaklik);
  Serial.print("°C  Hissi: ");
  Serial.print(hissi);
  Serial.println("°C");
}

💻 Arduino Kodu (İleri Versiyon - Veri Günlüğü ve Alarm)

#include 

#define DHTPIN 2
#define DHTTYPE DHT11
#define BUZZER_PIN 8
#define LED_PIN 13

DHT dht(DHTPIN, DHTTYPE);

// Veri depolama (son 10 ölçüm)
float sicakliklar[10];
float nemler[10];
int veri_sayisi = 0;
unsigned long son_okuma = 0;
const unsigned long OKUMA_ARALI = 2000; // 2 saniye

// Sıcaklık sınırları
const float MIN_SICAKLIK = 15.0;  // Minimum
const float MAX_SICAKLIK = 35.0;  // Maksimum
const float MIN_NEM = 30.0;       // Minimum
const float MAX_NEM = 80.0;       // Maksimum

void setup() {
  Serial.begin(9600);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);
  
  Serial.println("\n=== DHT11 İleri İzleme Sistemi ===");
  Serial.print("Sıcaklık Aralığı: ");
  Serial.print(MIN_SICAKLIK);
  Serial.print("°C - ");
  Serial.print(MAX_SICAKLIK);
  Serial.println("°C");
  
  Serial.print("Nem Aralığı: ");
  Serial.print(MIN_NEM);
  Serial.print("% - ");
  Serial.print(MAX_NEM);
  Serial.println("%\n");
  
  dht.begin();
}

void loop() {
  unsigned long simdi = millis();
  
  // Okuma zaman aralığı kontrol et
  if (simdi - son_okuma < OKUMA_ARALI) {
    return;
  }
  son_okuma = simdi;
  
  // DHT11'den oku
  float nem = dht.readHumidity();
  float sicaklik = dht.readTemperature();
  
  // Hata kontrol
  if (isnan(nem) || isnan(sicaklik)) {
    Serial.println("[HATA] DHT11 sensörü okunamadı!");
    digitalWrite(LED_PIN, HIGH);
    delay(200);
    digitalWrite(LED_PIN, LOW);
    return;
  }
  
  // Veri depoya ekle (son 10 ölçüm)
  if (veri_sayisi < 10) {
    sicakliklar[veri_sayisi] = sicaklik;
    nemler[veri_sayisi] = nem;
    veri_sayisi++;
  } else {
    // Eski verileri kaydır
    for (int i = 0; i < 9; i++) {
      sicakliklar[i] = sicakliklar[i + 1];
      nemler[i] = nemler[i + 1];
    }
    sicakliklar[9] = sicaklik;
    nemler[9] = nem;
  }
  
  // Hissi sıcaklık
  float hissi = dht.computeHeatIndex(sicaklik, nem, false);
  
  // İstatistikler
  float avg_sicaklik = hesaplaOrtalama(sicakliklar, veri_sayisi);
  float avg_nem = hesaplaOrtalama(nemler, veri_sayisi);
  float min_sicaklik = hesaplaMin(sicakliklar, veri_sayisi);
  float max_sicaklik = hesaplaMax(sicakliklar, veri_sayisi);
  
  // Konsola yaz
  Serial.print("[";
  Serial.print(simdi / 1000);
  Serial.print("s] ");
  Serial.print("Sıcaklık: ");
  Serial.print(sicaklik, 1);
  Serial.print("°C | Nem: ");
  Serial.print(nem, 1);
  Serial.print("% | Hissi: ");
  Serial.print(hissi, 1);
  Serial.println("°C");
  
  // Alarm kontrolü
  bool alarm_durumu = false;
  
  if (sicaklik < MIN_SICAKLIK) {
    Serial.println("⚠️  UYARI: Sıcaklık çok düşük!");
    alarm_durumu = true;
  }
  else if (sicaklik > MAX_SICAKLIK) {
    Serial.println("⚠️  UYARI: Sıcaklık çok yüksek!");
    alarm_durumu = true;
  }
  
  if (nem < MIN_NEM) {
    Serial.println("⚠️  UYARI: Nem çok düşük!");
    alarm_durumu = true;
  }
  else if (nem > MAX_NEM) {
    Serial.println("⚠️  UYARI: Nem çok yüksek!");
    alarm_durumu = true;
  }
  
  // Alarm sesini çal
  if (alarm_durumu) {
    tone(BUZZER_PIN, 1000, 200);
    delay(100);
    noTone(BUZZER_PIN);
  }
  
  // LED durumunu güncelle
  digitalWrite(LED_PIN, alarm_durumu ? HIGH : LOW);
  
  // İstatistikler (her 10 ölçümde)
  static int sayac = 0;
  sayac++;
  if (sayac >= 10) {
    Serial.println("\n=== İstatistikler ===");
    Serial.print("Ort. Sıcaklık: ");
    Serial.print(avg_sicaklik, 1);
    Serial.print("°C | Min: ");
    Serial.print(min_sicaklik, 1);
    Serial.print("°C | Max: ");
    Serial.print(max_sicaklik, 1);
    Serial.println("°C");
    
    Serial.print("Ort. Nem: ");
    Serial.print(avg_nem, 1);
    Serial.println("%\n");
    sayac = 0;
  }
}

// Ortalama hesapla
float hesaplaOrtalama(float dizi[], int boyut) {
  if (boyut == 0) return 0;
  float toplam = 0;
  for (int i = 0; i < boyut; i++) {
    toplam += dizi[i];
  }
  return toplam / boyut;
}

// Minimum değeri bul
float hesaplaMin(float dizi[], int boyut) {
  if (boyut == 0) return 0;
  float min_val = dizi[0];
  for (int i = 1; i < boyut; i++) {
    if (dizi[i] < min_val) min_val = dizi[i];
  }
  return min_val;
}

// Maksimum değeri bul
float hesaplaMax(float dizi[], int boyut) {
  if (boyut == 0) return 0;
  float max_val = dizi[0];
  for (int i = 1; i < boyut; i++) {
    if (dizi[i] > max_val) max_val = dizi[i];
  }
  return max_val;
}

🔍 Kod Açıklaması

readHumidity() ve readTemperature()

DHT11'den nem ve sıcaklık değerlerini okur. isnan() ile okumanın başarılı olup olmadığını kontrol ederiz.

computeHeatIndex()

Sıcaklık ve nem oranından "hissi sıcaklık" hesaplar. İnsanın algıladığı sıcaklıktır.

Veri Depolama (Circular Buffer)

Son 10 ölçümü saklar. Yeni ölçüm gelince en eski silinir. İstatistik hesaplanması için kullanılır.

Alarm Sistemi

Sıcaklık ve nem sınırlarından çıkarsa buzzer çalar ve LED yanar.

İstatistik Fonksiyonları

hesaplaOrtalama(), hesaplaMin(), hesaplaMax() - Veri analizi için fonksiyonlar.

🛠️ Sorun Giderme

❌ DHT11 okunamadı hatası

✅ Çözümler:

  • Bağlantıları kontrol et (VCC-5V, GND-GND, DATA-Pin2)
  • Pull-up direnci (10kΩ) veri hattına bağlı mı?
  • Sensör çok hızlı okumaya çalışma (minimum 2 saniye arası)
  • Jumper kablolar sıkı mı bağlı?

❌ Değerler sabit veya değişmiyor

✅ Çözümler:

  • DHT11'i bir dakika ısınması için bekle
  • Sensörü döndür veya elle ısında/soğutta deneyimle değeri kontrol et
  • Kütüphane düzgün kurulmuş mu? (Adafruit DHT ve Unified Sensor)

❌ Hissi sıcaklık garip gösteriliyorsa

✅ Çözümler:

  • Nem değeri tutarlı mı? (30-80% ideal)
  • Sıcaklık gerçekçi mi? (0-50°C normal aralık)
  • Sensörü doğru konuma yerleştir (işığın altında değil)
